cmake_minimum_required(VERSION 3.20)
project(BareClone VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(BareClone WIN32 main.cpp app.rc)
target_include_directories(BareClone PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(BareClone PRIVATE comctl32 ole32 uuid shell32 comdlg32 gdi32 uxtheme dwmapi)
if(MINGW)
 target_link_options(BareClone PRIVATE -municode -static -static-libgcc -static-libstdc++)
else()
 target_compile_options(BareClone PRIVATE /utf-8 /W4)
 set_property(TARGET BareClone PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_executable(search-tests tests.cpp)
add_test(NAME search-engine COMMAND search-tests)
